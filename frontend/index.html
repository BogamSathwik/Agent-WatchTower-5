🐍 Python Web App (Ask & Share)
Query Watchtower from Django, FastAPI, or Flask: 
import requests

# Ask Watchtower for agent fleet data via MCP
resp = requests.post("http://localhost:5000/api/v1/mcp", json={
    "jsonrpc": "2.0",
    "id": 1,
    "method": "tools/call",
    "params": {
        "name": "get_registered_agents",
        "arguments": { "limit": 10 }
    }
})
agents = resp.json()["result"]["structuredResult"]["agents"]
print(f"Fetched {len(agents)} agents from Watchtower")
